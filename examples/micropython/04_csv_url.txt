# label: Read a CSV from a URL
# MicroPython: a plain GET URL needs no directive — read it directly, same
# as plain pandas. The file is fetched in the browser and bound as a
# pandas_mpy DataFrame (no DuckDB needed for CSV — only parquet requires it).
# `ost.read(...)` exists for sources that need more than a URL (registries,
# secret_key, canonical queries, POST bodies, databases).
import pandas_mpy as pd
import plotly_express_mpy as pe

iris = pd.read_csv("https://raw.githubusercontent.com/hmelberg/openstat/main/data/iris.csv")

# The frame is ready under the name you chose:
show(iris.head())

# A summary table — pick columns, then group:
show(iris[["species", "sepal_length", "petal_length"]].groupby("species").mean())

# An interactive scatter plot:
pe.scatter(iris, x="sepal_length", y="petal_length", color="species",
           title="Iris — sepal vs petal")
